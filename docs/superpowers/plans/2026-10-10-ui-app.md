# UI 应用（ui.exe）Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 构建 `ui.exe` —— 录屏软件的控制界面与 IPC 客户端：托盘图标 + 录制时悬浮条（计时/FPS/停止）+ 轻量设置对话框 + 全局热键 + 区域框选，通过命名管道驱动已完成的 `recorder.exe`。

**Architecture:** `ui.exe` 是长驻进程，拥有托盘/热键/悬浮条/设置窗；录屏时按需启动 `recorder.exe --pipe <name>`，作为**管道客户端**连上（`recorder.exe` 是服务端，见 §IPC 约定），下发 `START`/`STOP`/`PING`，接收 `Ready`/`State`/`Stats`/`Error`/`Done`。UI 只做控制与展示，录制全部在子进程。

**Tech Stack:** C++17 / MSVC / 纯 Win32（无框架）+ `.rc` 对话框 + `Shell_NotifyIcon` 托盘 + `RegisterHotKey`；命名管道 IPC；nlohmann/json（经 `common`）。

**对应规格：** `docs/superpowers/specs/2026-10-10-screen-recorder-design.md`（§3 生命周期、§4 IPC、§5 组件）
**前置：** Plan 1（`recorder.exe`）已完成并合入 `master`。引擎单测 6/6 通过。

**IPC 约定（与已实现引擎一致，已核对 `recorder/main.cpp`）：**
- **`recorder.exe` = 命名管道服务端**：启动后 `CreateNamedPipe` 并 `ConnectNamedPipe` 等待客户端，连上先发 `Ready {"version":1}`。
- **`ui.exe` = 客户端**：`CreateFile(pipe)` 连接（recorder 可能还没建好管道 → 需重试）。
- 客户端 → 服务端：`Start <config-json>`、`Stop`、`Ping`。
- 服务端 → 客户端：`Ready`、`State {"path":...}`（启动成功）、`State "alive"`（Ping 应答）、`Stats {...}`（录制中每秒一条，**本次 Plan 新增**）、`Error {"code":"start_failed"}`、`Done {"ok":true}`。

```cpp
RecordConfig config_from_json(const std::string& body) {
  try {
    json j = json::parse(body);
    RecordConfig c;
    c.output_dir = j.value("output_dir", c.output_dir);
    c.monitor_index = j.value("monitor_index", c.monitor_index);
    c.fps = std::clamp(j.value("fps", c.fps), 1, 240);
    c.quality = quality_from_str(j.value("quality", std::string("medium")));

    c.region_enabled = j.value("region_enabled", c.region_enabled);
    c.include_cursor = j.value("include_cursor", c.include_cursor);
    if (j.contains("region")) {
      c.region.x = j["region"].value("x", c.region.x);
      c.region.y = j["region"].value("y", c.region.y);
      c.region.w = j["region"].value("w", c.region.w);
      c.region.h = j["region"].value("h", c.region.h);
    }
    if (j.contains("system_audio")) {
      c.system_audio.enabled = j["system_audio"].value("enabled", c.system_audio.enabled);
      c.system_audio.volume = j["system_audio"].value("volume", c.system_audio.volume);
      c.system_audio.device = j["system_audio"].value("device", c.system_audio.device);
    }
    if (j.contains("microphone")) {
      c.microphone.enabled = j["microphone"].value("enabled", c.microphone.enabled);
      c.microphone.volume = j["microphone"].value("volume", c.microphone.volume);
      c.microphone.device = j["microphone"].value("device", c.microphone.device);
```

```cpp
static int run_ipc(const std::string& pipe_name) {
  IpcServer server;
  if (!server.create(pipe_name)) { LOGE("pipe create failed"); return 3; }
  LOGI("waiting for UI on %s", pipe_name.c_str());
  if (!server.wait_client()) { LOGE("connect failed"); return 3; }
  server.send_message({MsgType::Ready, R"({"version":1})"});

  Pipeline p;
  std::atomic<bool> recording{false};
  std::thread stats_t;
  auto join_stats = [&] { if (stats_t.joinable()) stats_t.join(); };

  for (;;) {
    Message m;
    if (!server.read_message(m)) { LOGI("pipe closed"); break; }
    if (m.type == MsgType::Start) {
      RecordConfig cfg = config_from_json(m.payload);
      std::string path;
      if (p.start(cfg, path)) {
        json j; j["path"] = path;
        server.send_message({MsgType::State, j.dump()});
        recording = true;
        stats_t = std::thread([&] {
          while (recording) {
            Sleep(1000);
            if (!recording) break;
            Stats s = p.stats();
            json sj;
            sj["fps"] = s.fps;
            sj["elapsed_ms"] = s.elapsed_ms;
            sj["dropped_frames"] = s.dropped_frames;
            server.send_message({MsgType::Stats, sj.dump()});
          }
        });
      } else {
        server.send_message({MsgType::Error, R"({"code":"start_failed"})"});
      }
    } else if (m.type == MsgType::Stop) {
      p.stop();
      recording = false;
      join_stats();
      server.send_message({MsgType::Done, R"({"ok":true})"});
    } else if (m.type == MsgType::Ping) {
      server.send_message({MsgType::State, R"("alive")"});
    }
  }
  recording = false;
  join_stats();
  p.stop();
  server.close();
  return 0;
}
```

```cpp
#include "ui/ipc_client.h"
#include <windows.h>

bool IpcClient::connect(const std::string& pipe_name, int retries, int delay_ms) {
  for (int i = 0; i < retries; ++i) {
    HANDLE h = CreateFileA(pipe_name.c_str(), GENERIC_READ | GENERIC_WRITE, 0,
                           nullptr, OPEN_EXISTING, 0, nullptr);
    if (h != INVALID_HANDLE_VALUE) {
      DWORD mode = PIPE_READMODE_MESSAGE;
      SetNamedPipeHandleState(h, &mode, nullptr, nullptr);
      handle_ = h;
      return true;
    }
    Sleep(delay_ms);
  }
  return false;
}

bool IpcClient::send_message(const Message& m) {
  if (!handle_) return false;
  HANDLE h = static_cast<HANDLE>(handle_);
  auto buf = encode_message(m);
  DWORD wrote = 0;
  return WriteFile(h, buf.data(), (DWORD)buf.size(), &wrote, nullptr) != 0;
}

bool IpcClient::read_message(Message& out) {
  if (!handle_) return false;
  HANDLE h = static_cast<HANDLE>(handle_);
  for (;;) {
    size_t consumed = 0;
    if (decode_message(buf_.data(), buf_.size(), out, consumed)) {
      buf_.erase(buf_.begin(), buf_.begin() + consumed);
      return true;
    }
    uint8_t tmp[4096];
    DWORD got = 0;
    BOOL ok = ReadFile(h, tmp, sizeof(tmp), &got, nullptr);
    if (got > 0) buf_.insert(buf_.end(), tmp, tmp + got);
    if (ok) {
      if (got == 0) return false;
      continue;
    }
    if (GetLastError() == ERROR_MORE_DATA) continue;
    return false;
  }
}

void IpcClient::close() {
  if (handle_) {
    CloseHandle(static_cast<HANDLE>(handle_));
    handle_ = nullptr;
  }
}
```

```cpp
#include "ui/recorder_process.h"
#include "common/log.h"
#include <windows.h>
#include <vector>

void* spawn_recorder(const std::string& exe_path, const std::string& pipe_name) {
  std::string cmd = "\"" + exe_path + "\" --pipe " + pipe_name;
  std::vector<char> buf(cmd.begin(), cmd.end());
  buf.push_back('\0');
  STARTUPINFOA si = {sizeof(si)};
  PROCESS_INFORMATION pi = {};
  if (!CreateProcessA(nullptr, buf.data(), nullptr, nullptr, FALSE,
                      CREATE_NO_WINDOW, nullptr, nullptr, &si, &pi)) {
    LOGE("CreateProcess failed: %lu", GetLastError());
    return nullptr;
  }
  CloseHandle(pi.hThread);
  return pi.hProcess;
}

void kill_recorder(void* process) {
  HANDLE h = static_cast<HANDLE>(process);
  if (!h) return;
  if (WaitForSingleObject(h, 3000) == WAIT_TIMEOUT) {
    TerminateProcess(h, 0);
    WaitForSingleObject(h, 1000);
  }
  CloseHandle(h);
}
```

```cpp
#include "ui/region_selector.h"
#include <windows.h>

namespace {
RECT g_start = {};
RECT g_cur = {};
bool g_dragging = false;
bool g_ok = false;

LRESULT CALLBACK proc(HWND h, UINT msg, WPARAM wp, LPARAM lp) {
  switch (msg) {
    case WM_LBUTTONDOWN:
      g_start.left = g_cur.left = GET_X_LPARAM(lp);
      g_start.top = g_cur.top = GET_Y_LPARAM(lp);
      g_dragging = true;
      return 0;
    case WM_MOUSEMOVE:
      if (g_dragging) {
        g_cur.left = GET_X_LPARAM(lp);
        g_cur.top = GET_Y_LPARAM(lp);
        InvalidateRect(h, nullptr, TRUE);
      }
      return 0;
    case WM_LBUTTONUP:
      if (g_dragging) {
        g_cur.left = GET_X_LPARAM(lp);
        g_cur.top = GET_Y_LPARAM(lp);
        g_dragging = false;
        g_ok = true;
        DestroyWindow(h);
      }
      return 0;
    case WM_RBUTTONUP:
    case WM_KEYDOWN:
      if (msg == WM_KEYDOWN && wp != VK_ESCAPE) return 0;
      DestroyWindow(h);
      return 0;
    case WM_PAINT: {
      PAINTSTRUCT ps;
      HDC dc = BeginPaint(h, &ps);
      RECT client;
      GetClientRect(h, &client);
      HBRUSH b = CreateSolidBrush(RGB(0, 0, 0));
      FillRect(dc, &client, b);
      DeleteObject(b);
      HPEN pen = CreatePen(PS_SOLID, 2, RGB(255, 40, 40));
      HGDIOBJ oldp = SelectObject(dc, pen);
      HGDIOBJ oldb = SelectObject(dc, GetStockObject(NULL_BRUSH));
      Rectangle(dc, g_start.left, g_start.top, g_cur.left, g_cur.top);
      SelectObject(dc, oldb);
      SelectObject(dc, oldp);
      DeleteObject(pen);
      EndPaint(h, &ps);
      return 0;
    }
    case WM_DESTROY:
      PostQuitMessage(0);
      return 0;
  }
  return DefWindowProc(h, msg, wp, lp);
}
}  // namespace

bool pick_region(HWND owner, RECT& out) {
  (void)owner;
  HINSTANCE inst = GetModuleHandleA(nullptr);
  WNDCLASSA wc = {};
  wc.lpfnWndProc = proc;
  wc.hInstance = inst;
  wc.lpszClassName = "ScreenRecRegionClass";
  wc.hCursor = LoadCursor(nullptr, IDC_CROSS);
  RegisterClassA(&wc);

  int x = GetSystemMetrics(SM_XVIRTUALSCREEN);
  int y = GetSystemMetrics(SM_YVIRTUALSCREEN);
  int w = GetSystemMetrics(SM_CXVIRTUALSCREEN);
  int h = GetSystemMetrics(SM_CYVIRTUALSCREEN);
  // WS_EX_LAYERED 必须在创建时就带上，否则随后的 SetLayeredWindowAttributes 失效，覆盖层会不透明遮住屏幕。
  HWND hwnd = CreateWindowExA(WS_EX_TOPMOST | WS_EX_TOOLWINDOW | WS_EX_LAYERED,
                              wc.lpszClassName, "", WS_POPUP, x, y, w, h,
                              nullptr, nullptr, inst, nullptr);
  if (!hwnd) return false;
  SetLayeredWindowAttributes(hwnd, 0, 90, LWA_ALPHA);
  ShowWindow(hwnd, SW_SHOW);
  SetForegroundWindow(hwnd);

  g_ok = false;
  MSG msg;
  while (GetMessage(&msg, nullptr, 0, 0) > 0) {
    TranslateMessage(&msg);
    DispatchMessage(&msg);
  }
  if (!g_ok) return false;
  RECT r;
  r.left = g_start.left + x; r.top = g_start.top + y;
  r.right = g_cur.left + x;  r.bottom = g_cur.top + y;
  if (r.right < r.left) { LONG t = r.left; r.left = r.right; r.right = t; }
  if (r.bottom < r.top) { LONG t = r.top; r.top = r.bottom; r.bottom = t; }
  if (r.right - r.left < 8 || r.bottom - r.top < 8) return false;
  out = r;
  return true;
}
```

```cpp
LRESULT CALLBACK Toolbar::wnd_proc(HWND h, UINT m, WPARAM w, LPARAM l) {
  Toolbar* self = (Toolbar*)GetWindowLongPtrA(h, GWLP_USERDATA);
  if (m == WM_NCCREATE) {
    CREATESTRUCTA* cs = (CREATESTRUCTA*)l;
    self = (Toolbar*)cs->lpCreateParams;
    SetWindowLongPtrA(h, GWLP_USERDATA, (LONG_PTR)self);
    self->hwnd_ = h;

  if (self) return self->handle(h, m, w, l);
  return DefWindowProc(h, m, w, l);
}

LRESULT Toolbar::handle(HWND h, UINT m, WPARAM w, LPARAM l) {
  switch (m) {
    case WM_CREATE: {
      CreateWindowExA(0, "STATIC", "REC", WS_CHILD | WS_VISIBLE | SS_CENTER,
                      6, 12, 34, 16, h, nullptr, nullptr, nullptr);
      CreateWindowExA(0, "STATIC", "00:00  30 fps", WS_CHILD | WS_VISIBLE,

      return 0;

      if (r == HTCLIENT) return HTCLIENT;  // 让子控件收到点击
      return r;
    }
    case WM_CLOSE:
      return 0;  // 由 stop 流程关闭
  }
  return DefWindowProc(h, m, w, l);
}
```

```cpp
#include "ui/app.h"
#include "common/settings.h"
#include "common/log.h"
#include "ui/recorder_process.h"
#include "ui/settings_dialog.h"
#include "ui/hotkey.h"
#include <nlohmann/json.hpp>

using nlohmann::json;

static std::string exe_dir() {
  char buf[MAX_PATH];
  GetModuleFileNameA(nullptr, buf, MAX_PATH);
  std::string s = buf;
  size_t p = s.find_last_of("\\/");
  return p == std::string::npos ? "." : s.substr(0, p);
}

bool App::init(HINSTANCE inst) {
  inst_ = inst;
  cfg_ = load_settings(default_settings_path());
  if (cfg_.output_dir.empty()) {
    char buf[MAX_PATH];
    if (SUCCEEDED(SHGetFolderPathA(nullptr, CSIDL_MYVIDEO, nullptr, 0, buf)))
      cfg_.output_dir = buf;
    else
      cfg_.output_dir = ".";
  }

  WNDCLASSA wc = {};
  wc.lpfnWndProc = [](HWND h, UINT m, WPARAM w, LPARAM l) -> LRESULT {
    App* self = (App*)GetWindowLongPtrA(h, GWLP_USERDATA);
    if (m == WM_NCCREATE) {
      self = (App*)((CREATESTRUCTA*)l)->lpCreateParams;
      SetWindowLongPtrA(h, GWLP_USERDATA, (LONG_PTR)self);
    }
    if (self) { self->on_message(h, m, w, l); return 0; }
    return DefWindowProc(h, m, w, l);
  };
  wc.hInstance = inst;
  wc.lpszClassName = "ScreenRecUiClass";
  RegisterClassA(&wc);
  hwnd_ = CreateWindowExA(0, wc.lpszClassName, "ScreenRecorder", 0, 0, 0, 0, 0,
                          nullptr, nullptr, inst, this);

  toolbar_.create(inst, [this] { stop_recording(); });
  tray_.add(hwnd_, WM_TRAY);
  tray_.set_callbacks([this] { toggle(); }, [this] { open_settings(); },
                      [this] { quit(); });
  register_toggle_hotkey(hwnd_, VK_F9);
  LOGI("ui ready, output -> %s", cfg_.output_dir.c_str());
  return hwnd_ != nullptr;
}

bool App::start_recording() {
  if (recording_) return false;
  pipe_name_ = "\\\\.\\pipe\\screenrec-default";
  proc_ = spawn_recorder(exe_dir() + "\\recorder.exe", pipe_name_);
  if (!proc_) { MessageBoxA(hwnd_, "启动录制进程失败", "错误", MB_OK | MB_ICONERROR); return false; }
  if (!client_.connect(pipe_name_, 60, 50)) {
    MessageBoxA(hwnd_, "连接录制进程失败", "错误", MB_OK | MB_ICONERROR);
    kill_recorder(proc_); proc_ = nullptr;
    return false;
  }
  Message ready;
  client_.read_message(ready);  // Ready
  client_.send_message({MsgType::Start, config_to_json(cfg_)});

  recording_ = true;
  reader_run_ = true;
  reader_t_ = std::thread([this] { reader_loop(); });
  toolbar_.show();
  return true;
}

void App::stop_recording() {
  if (!recording_) return;
  client_.send_message({MsgType::Stop, ""});
  // reader 线程收到 Done 后收尾
}

void App::reader_loop() {
  while (reader_run_) {
    Message m;
    if (!client_.read_message(m)) break;
    if (m.type == MsgType::State) {
      LOGI("state: %s", m.payload.c_str());
    } else if (m.type == MsgType::Stats) {
      try {
        json j = json::parse(m.payload);
        toolbar_.set_status(j.value("elapsed_ms", (int64_t)0), j.value("fps", 0.0));
      } catch (...) {}
    } else if (m.type == MsgType::Error) {
      MessageBoxA(hwnd_, "录制出错", "错误", MB_OK | MB_ICONERROR);
    } else if (m.type == MsgType::Done) {
      break;
    }
  }
  recording_ = false;
  reader_run_ = false;
  client_.close();
  PostMessageA(hwnd_, WM_APP + 2, 0, 0);  // 通知主线程收尾
}

void App::toggle() { if (recording_) stop_recording(); else start_recording(); }

void App::open_settings() {
  if (recording_) { MessageBoxA(hwnd_, "录制中不能改设置", "提示", MB_OK); return; }
  if (show_settings_dialog(hwnd_, cfg_)) {
    save_settings(default_settings_path(), cfg_);
  }
}

void App::quit() {
  if (recording_) stop_recording();
  if (reader_t_.joinable()) reader_t_.join();
  kill_recorder(proc_); proc_ = nullptr;
  tray_.remove();
  unregister_toggle_hotkey(hwnd_);
  PostQuitMessage(0);
}

void App::on_message(HWND h, UINT msg, WPARAM wp, LPARAM lp) {
  switch (msg) {
    case WM_TRAY:
      if (lp == WM_RBUTTONUP) tray_.show_menu(h);
      else if (lp == WM_LBUTTONDBLCLK) toggle();
      break;
    case WM_COMMAND:
      tray_.on_command(h, LOWORD(wp));
      break;
    case WM_HOTKEY:
      if (wp == 1) toggle();
      break;
    case WM_APP + 2:  // reader 结束 → 收尾
      if (reader_t_.joinable()) reader_t_.join();
      kill_recorder(proc_); proc_ = nullptr;
      toolbar_.hide();
      break;
  }
}
```
